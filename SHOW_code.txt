N_PERM = 10000       
SEED = 42

np.random.seed(SEED)
rng = np.random.default_rng(SEED)

def permutation_test(cluster_values, labels, n_perm=N_PERM, blocks=None):
    labels = np.array(labels)
    cluster_values = np.array(cluster_values, dtype=float)

    if np.isnan(cluster_values).any():
        return np.nan, np.nan, np.array([])

    observed = (cluster_values[labels == "treated"].mean() - cluster_values[labels == "control"].mean())
    perm_stats = np.empty(n_perm)
    for i in range(n_perm):
        if blocks is None:
            perm_labels = rng.permutation(labels)
        else:
            perm_labels = labels.copy()
            for b in pd.unique(blocks):
                m = blocks == b
                perm_labels[m] = rng.permutation(labels[m])
        perm_stats[i] = (cluster_values[perm_labels == "treated"].mean() - cluster_values[perm_labels == "control"].mean())
    r = int((np.abs(perm_stats) >= np.abs(observed)).sum())
    p_value = (1 + r) / (1 + n_perm)
    return observed, p_value, perm_stats

def benjamini_hochberg(p):
    p = np.asarray(p, dtype=float)
    n = len(p)
    q = np.empty(n)
    prev = 1.0
    for rank, i in enumerate(np.argsort(p)[::-1]):
        prev = min(prev, p[i] * n / (n - rank))
        q[i] = prev
    return q

def load_strata(soft_path, subset_type):
    if not os.path.exists(soft_path):
        print(f"    [주의] {soft_path} 없음 -> 계층 비교 생략")
        return None
    import gzip
    mapping, cur = {}, {}
    with gzip.open(soft_path, "rt", errors="ignore") as fh:
        for line in fh:
            if line.startswith("^SUBSET"):
                cur = {}
            elif line.startswith("!subset_description"):
                cur["desc"] = line.split("=", 1)[1].strip()
            elif line.startswith("!subset_sample_id"):
                cur["ids"] = [s.strip() for s in line.split("=", 1)[1].split(",")]
            elif line.startswith("!subset_type"):
                if line.split("=", 1)[1].strip() == subset_type:
                    for s in cur["ids"]:
                        mapping[s] = cur["desc"]
            elif line.startswith("!dataset_table_begin"):
                break
    return mapping or None

cluster_stats = []
null_dists = {}

    obs, pval, perm_stats = permutation_test(values, labels, blocks=None)
    n_genes_c = int((gene_cluster_df["cluster"] == int(cluster.split("_")[1])).sum())
    row = {"cluster": cluster, "n_genes": n_genes_c,
           "delta_mean (T-C)": obs, "p_value": pval,
           "r": int((np.abs(perm_stats) >= abs(obs)).sum()) if len(perm_stats) else np.nan,
           "n_perm": N_PERM,
           "abs_percentile": (float((np.abs(perm_stats) < abs(obs)).mean() * 100) if len(perm_stats) else np.nan),
           "null_sd": perm_stats.std() if len(perm_stats) else np.nan}

        row["null_sd_ratio"] = row["null_sd"] / strat_stats.std()
        null_dists[f"{cluster}__strat"] = strat_stats
    cluster_stats.append(row)
